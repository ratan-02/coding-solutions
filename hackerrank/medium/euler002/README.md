```cpp
#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;

    while (t--) {
        long long n;
        cin >> n;

        long long sum = 0;
        long long a = 2, b = 8;

        while (a <= n) {
            sum += a;
            long long c = 4 * b + a;
            a = b;
            b = c;
        }

        cout << sum << endl;
    }

    return 0;
}
```
